layout LayeredCounter
  node _T kind=op pins=_T~in0:WEST,_T~in1:WEST
  node _T_1 kind=op pins=_T_1~in0:WEST,_T_1~in1:WEST
  node _T_2 kind=op pins=_T_2~in0:WEST,_T_2~in1:WEST
  node _T_3 kind=op pins=_T_3~in0:WEST,_T_3~in1:WEST
  node _T_4 kind=op pins=_T_4~in0:WEST,_T_4~in1:WEST
  node _count_T kind=op pins=_count_T:EAST,_count_T~in0:WEST,_count_T~in1:WEST
  node _count_T_2 kind=op pins=_count_T_2:EAST,_count_T_2~in0:WEST,_count_T_2~in1:WEST,_count_T_2~in2:WEST
  node const$_T_1~in1 kind=const pins=const$_T_1~in1:EAST
  node const$_T_2~in1 kind=const pins=const$_T_2~in1:EAST
  node const$_T_3~in1 kind=const pins=const$_T_3~in1:EAST
  node const$_T_4~in1 kind=const pins=const$_T_4~in1:EAST
  node const$_count_T_2~in1 kind=const pins=const$_count_T_2~in1:EAST
  node const$_count_T~in1 kind=const pins=const$_count_T~in1:EAST
  node count kind=reg pins=count:WEST,count~q:EAST
  node io.count kind=boundary pins=io.count:WEST:iface
  node io.en kind=boundary pins=io.en:EAST:iface
  node io.limit kind=boundary pins=io.limit:EAST:iface
  node io.wrap kind=boundary pins=io.wrap:WEST:iface
  node mux$count kind=op pins=mux$count:EAST,mux$count~in0:WEST,mux$count~in1:WEST,mux$count~in2:WEST
  node reset kind=boundary pins=reset:EAST
  node wrap kind=op pins=wrap:EAST,wrap~in0:WEST,wrap~in1:WEST
  edge _T_1~in0 -> reset
  edge _T_1~in1 -> const$_T_1~in1
  edge _T_2~in0 -> io.limit
  edge _T_2~in1 -> const$_T_2~in1
  edge _T_3~in0 -> reset
  edge _T_3~in1 -> const$_T_3~in1
  edge _T_4~in0 -> reset
  edge _T_4~in1 -> const$_T_4~in1
  edge _T~in0 -> count~q
  edge _T~in1 -> io.limit
  edge _count_T_2~in0 -> wrap
  edge _count_T_2~in1 -> const$_count_T_2~in1
  edge _count_T_2~in2 -> _count_T
  edge _count_T~in0 -> count~q
  edge _count_T~in1 -> const$_count_T~in1
  edge count -> mux$count
  edge io.count -> count~q
  edge io.wrap -> wrap
  edge mux$count~in0 -> io.en
  edge mux$count~in1 -> _count_T_2
  edge mux$count~in2 -> count~q
  edge wrap~in0 -> count~q
  edge wrap~in1 -> io.limit
